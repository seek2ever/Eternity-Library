阶段 1：统一数据库读取工作线程

适用项目：Eternity-Library（三木书斋）　前置条件：阶段 0 已完成

本指南的代码可以直接粘贴。代码按当前仓库的 book_record.py、 database.py、el_gui.py 组织方式编写，不是伪代码。本文只迁移 “读取”操作；扫描入库仍暂时使用主线程中的 DatabaseManager，这是阶段 1 的明确过渡边界。不要在本阶段删除 DatabaseManager。

1. 阶段 1 完成后的结构

主线程只发出读取请求并更新 Model；数据库连接在 DatabaseWorker 所在的 QThread 中创建、使用和关闭。Worker 不接触任何 QWidget，也不把 SQLite 连接传回主线程。

文件阶段 1 的处理
book_record.py保持不变，继续作为查询结果的统一数据结构。
database.py保留 DatabaseManager；新增并使用 DatabaseWorker。
el_gui.py移除 BookQueryWorker 的使用，改为统一 Worker 和请求信号。
books.py本阶段保持不变，仍由 DatabaseManager 负责写入。
verify_phase1.py新增读取 Worker 的独立验证脚本。
关键原则：不能把主线程已经创建的 sqlite3.Connection 传给 Worker。连接必须在 Worker 的 initialize() 槽中创建，并在 close_database() 槽中关闭。

2. 第一步：在 database.py 中新增 DatabaseWorker

打开项目根目录的 database.py。保留文件顶部现有导入、完整的 DatabaseManager，然后把下面这段完整代码粘贴到 DatabaseManager 类结束之后、原 BookQueryWorker 之前。

class DatabaseWorker(QObject):
    """在专用 QThread 中串行执行所有数据库读取操作。"""

    initialized = Signal()
    books_ready = Signal(list)
    query_error = Signal(str)
    database_closed = Signal()

    def __init__(self, db_name="books_information.db"):
        super().__init__()
        self.db_name = db_name
        self.connection: sqlite3.Connection | None = None

    @Slot()
    def initialize(self):
        """必须由 Worker 所在线程调用，创建该线程专属的连接。"""
        if self.connection is not None:
            return

        try:
            self.connection = sqlite3.connect(self.db_name)
            self.connection.execute(
                """
                CREATE TABLE IF NOT EXISTS books_information (
                    book_id INTEGER,
                    book_name TEXT NOT NULL,
                    book_path TEXT,
                    add_time TEXT,
                    author TEXT,
                    nationality TEXT,
                    translator TEXT,
                    publisher TEXT,
                    publication_date TEXT,
                    level TEXT,
                    read_status TEXT,
                    book_type TEXT,
                    isbn INTEGER,
                    pages INTEGER,
                    read_progress TEXT,
                    read_time REAL,
                    read_date TEXT,
                    read_link TEXT,
                    introduction TEXT
                )
                """
            )
            self.connection.commit()
        except sqlite3.Error as error:
            self._close_connection()
            self.query_error.emit(f"数据库初始化失败：{error}")
            return

        self.initialized.emit()

    def _close_connection(self):
        if self.connection is not None:
            self.connection.close()
            self.connection = None

    def _select(self, sql: str, params: tuple = ()) -> list[BookRecord] | None:
        """执行一次参数化查询，并把结果转换成阶段 0 的数据结构。"""
        if self.connection is None:
            self.query_error.emit("数据库尚未初始化，无法查询。")
            return None

        try:
            rows = self.connection.execute(sql, params).fetchall()
            return rows_to_records(rows)
        except sqlite3.Error as error:
            self.query_error.emit(f"数据库查询失败：{error}")
        except (TypeError, ValueError) as error:
            self.query_error.emit(f"书籍数据格式错误：{error}")
        return None

    @Slot()
    def fetch_all_books(self):
        """查询全部书籍；空数据库也会发送 books_ready([])。"""
        books = self._select(
            "SELECT * FROM books_information ORDER BY book_id"
        )
        if books is not None:
            self.books_ready.emit(books)

    @Slot(str)
    def fetch_books_by_type(self, book_type: str):
        """按书籍类型精确筛选。"""
        books = self._select(
            """
            SELECT * FROM books_information
            WHERE book_type = ?
            ORDER BY book_id
            """,
            (book_type,),
        )
        if books is not None:
            self.books_ready.emit(books)

    @Slot(str)
    def fetch_books_by_status(self, read_status: str):
        """按阅读状态精确筛选。"""
        books = self._select(
            """
            SELECT * FROM books_information
            WHERE read_status = ?
            ORDER BY book_id
            """,
            (read_status,),
        )
        if books is not None:
            self.books_ready.emit(books)

    @Slot(str)
    def fetch_books_by_author(self, author: str):
        """按作者精确筛选。"""
        books = self._select(
            """
            SELECT * FROM books_information
            WHERE author = ?
            ORDER BY book_id
            """,
            (author,),
        )
        if books is not None:
            self.books_ready.emit(books)

    @Slot()
    def close_database(self):
        """必须由 Worker 所在线程调用。关闭连接后通知主线程退出线程。"""
        self._close_connection()
        self.database_closed.emit()

4. 第三步：完整替换 _setup_book_worker

在 MainWindow 中找到原来的 _setup_book_worker， 用下面完整方法替换。它不会把连接传入线程；所有跨线程调用都通过信号排队。

def _setup_book_worker(self):
    """创建读取线程，并在 Worker 初始化完成后加载全部书籍。"""
    self._books_cache: list[BookRecord] | None = None
    self._cache_timestamp = 0.0

    self._database_thread = QThread(self)
    self._database_worker = DatabaseWorker()
    self._database_worker.moveToThread(self._database_thread)

    self._database_worker.initialized.connect(
        self._on_database_initialized
    )
    self._database_worker.books_ready.connect(self._on_books_loaded)
    self._database_worker.query_error.connect(self._on_query_error)
    self._database_worker.database_closed.connect(
        self._database_thread.quit
    )

    self.request_initialize.connect(
        self._database_worker.initialize
    )
    self.request_fetch_all.connect(
        self._database_worker.fetch_all_books
    )
    self.request_fetch_by_type.connect(
        self._database_worker.fetch_books_by_type
    )
    self.request_fetch_by_status.connect(
        self._database_worker.fetch_books_by_status
    )
    self.request_fetch_by_author.connect(
        self._database_worker.fetch_books_by_author
    )
    self.request_close_database.connect(
        self._database_worker.close_database
    )

    self._database_thread.finished.connect(
        self._database_worker.deleteLater
    )
    self._database_thread.start()
    self.request_initialize.emit()

然后在 MainWindow.__init__ 中保留原来的：

self._setup_book_worker()

不要再创建 self._book_worker = BookQueryWorker()，也不要再创建 第二个查询线程。

5. 第四步：替换查询触发和结果处理方法

把原来的 refresh_view、_on_query_error、 _on_books_loaded 替换成下面三个方法。这里保留 30 秒缓存，但缓存只 缓存“全部书籍”结果；筛选请求会直接走 Worker。

def refresh_view(self) -> None:
    """刷新当前数据；缓存有效时不重复查询全部书籍。"""
    if (
        self._books_cache is not None
        and (time.time() - self._cache_timestamp) < 30
    ):
        self._on_books_loaded(self._books_cache)
        return

    self.request_fetch_all.emit()

def _on_query_error(self, error_msg: str) -> None:
    """所有数据库读取错误都在主线程提示用户。"""
    Tips.information_msg(f"数据库查询失败：{error_msg}")
    if self.statusbar is not None:
        self.statusbar.showMessage("数据库查询失败")

def _on_database_initialized(self) -> None:
    """Worker 完成连接和建表后，才允许发出查询请求。"""
    if self.statusbar is not None:
        self.statusbar.showMessage("数据库已就绪")
    self.request_fetch_all.emit()

def _on_books_loaded(self, books: list[BookRecord]) -> None:
    """在主线程接收结果并刷新两个 Model。"""
    self._books_cache = books
    self._cache_timestamp = time.time()
    self.cover_model.set_books(books)
    self.table_model.set_data(books)

    if books:
        message = self._translate("Views", f"已刷新，共 {len(books)} 本书")
    else:
        message = self._translate("Views", "数据库中暂无书籍")
    if self.statusbar is not None:
        self.statusbar.showMessage(message)
为什么空列表也必须发信号：查询结果为空是成功结果，不是错误。 _on_books_loaded([]) 会清空两个 Model，避免用户删除或筛选后界面继续 显示旧数据。

7. 第六步：安全关闭线程

把 closeEvent 替换为下面的方法。顺序是：发出关闭请求 → Worker 关闭自己的连接 → 发出 database_closed → 线程退出 → 主线程等待结束。

def closeEvent(self, event) -> None:
    """窗口关闭时先关闭 Worker 连接，再退出线程。"""
    self.request_close_database.emit()
    if not self._database_thread.wait(3000):
        Tips.information_msg("数据库线程未能在 3 秒内正常退出。")
        event.ignore()
        return
    event.accept()
不要改成 terminate()。强制终止可能发生在 SQLite 正在读取的中间状态，也会留下连接或资源没有释放。wait() 返回 False 时保留窗口，用户可以看到问题，而不是静默退出。

9. 第八步：新增可直接运行的验证脚本

在项目根目录新建 verify_phase1.py，复制以下完整代码。它使用临时 数据库和真正的 QThread，验证初始化、全部查询、三种筛选、空结果和 关闭流程；不会修改项目中的 books_information.db。

from __future__ import annotations

import os
import sqlite3
import sys
import tempfile

from PySide6.QtCore import QCoreApplication, QObject, QThread, Signal

from book_record import BOOK_COLUMNS
from database import DatabaseWorker


class WorkerCheck(QObject):
    initialize_requested = Signal()
    fetch_all_requested = Signal()
    fetch_type_requested = Signal(str)
    fetch_status_requested = Signal(str)
    fetch_author_requested = Signal(str)
    close_requested = Signal()

    def __init__(self, app: QCoreApplication, db_name: str):
        super().__init__()
        self.app = app
        self.thread = QThread()
        self.worker = DatabaseWorker(db_name)
        self.worker.moveToThread(self.thread)
        self.initialize_requested.connect(self.worker.initialize)
        self.fetch_all_requested.connect(self.worker.fetch_all_books)
        self.fetch_type_requested.connect(self.worker.fetch_books_by_type)
        self.fetch_status_requested.connect(self.worker.fetch_books_by_status)
        self.fetch_author_requested.connect(self.worker.fetch_books_by_author)
        self.close_requested.connect(self.worker.close_database)
        self.worker.initialized.connect(self.on_initialized)
        self.worker.books_ready.connect(self.on_books_ready)
        self.worker.query_error.connect(self.on_error)
        self.worker.database_closed.connect(self.on_database_closed)
        self.thread.finished.connect(self.worker.deleteLater)
        self.step = 0
        self.expected = [
            ("all", 2),
            ("type", 1),
            ("status", 1),
            ("author", 1),
            ("empty", 0),
        ]
        self.thread.start()
        self.initialize_requested.emit()

    def on_initialized(self):
        self.fetch_all_requested.emit()

    def on_books_ready(self, books):
        name, expected_count = self.expected[self.step]
        actual_count = len(books)
        assert actual_count == expected_count, (
            f"{name} 查询应返回 {expected_count} 条，实际 {actual_count} 条"
        )
        if name == "all":
            assert len(books[0].to_row()) == len(BOOK_COLUMNS)
            self.fetch_type_requested.emit(".pdf")
        elif name == "type":
            self.fetch_status_requested.emit("正在阅读")
        elif name == "status":
            self.fetch_author_requested.emit("作者A")
        elif name == "author":
            self.fetch_type_requested.emit("不存在的类型")
        else:
            self.close_requested.emit()
        self.step += 1

    def on_error(self, message):
        raise RuntimeError(message)

    def on_database_closed(self):
        self.thread.quit()
        self.thread.finished.connect(self.app.quit)


def create_database(path: str) -> None:
    connection = sqlite3.connect(path)
    try:
        connection.execute(
            """
            CREATE TABLE books_information (
                book_id INTEGER, book_name TEXT NOT NULL, book_path TEXT,
                add_time TEXT, author TEXT, nationality TEXT, translator TEXT,
                publisher TEXT, publication_date TEXT, level TEXT,
                read_status TEXT, book_type TEXT, isbn INTEGER, pages INTEGER,
                read_progress TEXT, read_time REAL, read_date TEXT,
                read_link TEXT, introduction TEXT
            )
            """
        )
        values = [
            (1, "甲", "a", "", "作者A", "", "", "", "", "",
             "正在阅读", ".pdf", None, None, "", None, "", "", ""),
            (2, "乙", "b", "", "作者B", "", "", "", "", "",
             "尚未阅读", ".txt", None, None, "", None, "", "", ""),
        ]
        connection.executemany(
            "INSERT INTO books_information VALUES "
            "(?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
            values,
        )
        connection.commit()
    finally:
        connection.close()


def main() -> None:
    app = QCoreApplication(sys.argv)
    file_descriptor, db_name = tempfile.mkstemp(suffix=".db")
    os.close(file_descriptor)
    try:
        os.remove(db_name)
        create_database(db_name)
        check = WorkerCheck(app, db_name)
        exit_code = app.exec()
        assert check.step == len(check.expected), "Worker 验证没有完成"
        print("阶段 1 DatabaseWorker 检查全部通过")
        raise SystemExit(exit_code)
    finally:
        if os.path.exists(db_name):
            os.remove(db_name)


if __name__ == "__main__":
    main()

11. 阶段 1 完成检查表

  • 已新增一个长期运行的 DatabaseWorker，而不是每次查询创建线程。
  • SQLite 连接只在 Worker 线程中创建、使用和关闭。
  • 全部查询、按类型、按状态、按作者查询都通过信号槽执行。
  • 查询结果统一为 list[BookRecord]。
  • 空结果发送 books_ready([])，两个 Model 会被清空。
  • 查询失败发送 query_error(str)，不会把错误文本伪装成列表。
  • 主线程只更新缓存、Model、View 和提示框。
  • 写入仍由 DatabaseManager 负责，没有在阶段 1 混入批量写入重构。
  • 验证脚本和 GUI 都能运行，关闭时没有线程销毁警告。

12. 阶段 2 的入口

阶段 1 完成后不要继续扩展这个 Worker 的写入方法。下一阶段应单独设计 insert_many：扫描线程只负责产生 list[BookRecord]， Worker 在线程内开启一个事务，统一处理重复策略、插入、提交或回滚，完成后发出 变更结果信号。这样可以在不破坏阶段 1 读取边界的情况下迁移写入。
