3. database.py 改动 —— 新增 BookQueryWorker 类

为什么放在 database.py 而不是新建文件？

  • BookQueryWorker 是纯数据库操作层，与 DatabaseManager 共享 db_name，属于同一职责域
  • 新建 book_worker.py 会导致模块碎片化——这个类只有 35 行
  • 以后需要按类型/状态/作者查询时，在这个类中增加对应方法即可，职责内聚

完整代码（追加到 database.py 末尾，if __name__ == '__main__' 之前）

class BookQueryWorker(QObject):
    """在后台线程中执行数据库查询，避免阻塞 UI 主线程

    关键设计：
    - __init__ 只接收 db_name（字符串），不接收连接/游标对象
    - 每次查询在自己的线程内创建独立 sqlite3 连接，查完即关
    - 这是 SQLite 线程安全的基本要求：连接不能跨线程共享
    """
    books_ready = Signal(list)    # 查询成功，携带书籍列表
    query_error = Signal(str)     # 查询失败，携带错误信息

    def __init__(self, db_name: str):
        """
        :param db_name: 数据库文件名（如 'books_information.db'）
        """
        super().__init__()
        self.db_name = db_name

    def fetch_all_books(self):
        """查询全部书籍（在工作线程中执行）"""
        import sqlite3
        try:
            conn = sqlite3.connect(self.db_name)
            cursor = conn.cursor()
            cursor.execute("SELECT * FROM books_information")
            books = cursor.fetchall()
            cursor.close()
            conn.close()
            self.books_ready.emit(books)
        except Exception as e:
            self.query_error.emit(str(e))

    def fetch_books_by_type(self, book_type: str):
        """按类型查询（预留，待后续接入分类筛选功能）"""
        import sqlite3
        try:
            conn = sqlite3.connect(self.db_name)
            cursor = conn.cursor()
            cursor.execute("SELECT * FROM books_information WHERE book_type=?", (book_type,))
            books = cursor.fetchall()
            cursor.close()
            conn.close()
            self.books_ready.emit(books)
        except Exception as e:
            self.query_error.emit(str(e))

    def fetch_books_by_status(self, status: str):
        """按阅读状态查询（预留）"""
        import sqlite3
        try:
            conn = sqlite3.connect(self.db_name)
            cursor = conn.cursor()
            cursor.execute("SELECT * FROM books_information WHERE read_status=?", (status,))
            books = cursor.fetchall()
            cursor.close()
            conn.close()
            self.books_ready.emit(books)
        except Exception as e:
            self.query_error.emit(str(e))

    def fetch_books_by_author(self, author: str):
        """按作者查询（预留）"""
        import sqlite3
        try:
            conn = sqlite3.connect(self.db_name)
            cursor = conn.cursor()
            cursor.execute("SELECT * FROM books_information WHERE author=?", (author,))
            books = cursor.fetchall()
            cursor.close()
            conn.close()
            self.books_ready.emit(books)
        except Exception as e:
            self.query_error.emit(str(e))
注意：每次查询都新建 sqlite3.connect()，查完立即 close()。不能在 __init__ 中创建连接然后复用——SQLite 连接对象不是线程安全的，跨线程使用会导致不可预测的崩溃。
代码位置：追加在 DatabaseManager 类定义之后、if __name__ == '__main__' 之前。两个类共享同一个 sqlite3 导入。

4.3 新增方法：_setup_book_worker() （插入到 __init__ 和 setup_ui 之间）

这是线程初始化的核心方法，负责创建工作线程、移动 Worker、连接信号、准备缓存：

def _setup_book_worker(self):
    """初始化后台查询线程、缓存变量"""

    # --- 缓存 ---
    self._books_cache = None         # 缓存最近一次查询结果
    self._cache_timestamp = 0        # 缓存创建时间 (time.time())

    # --- 工作线程 ---
    self._query_thread = QThread()

    # 创建 Worker（传入数据库文件名，不传连接对象）
    self._book_worker = BookQueryWorker(self.db.db_name)

    # 把 Worker 移到工作线程（此后 Worker 的所有槽函数在工作线程执行）
    self._book_worker.moveToThread(self._query_thread)

    # 连接信号：Worker 查询完成 → 在主线程处理结果
    self._book_worker.books_ready.connect(self._on_books_loaded)
    self._book_worker.query_error.connect(self._on_query_error)

    # 启动线程（进入事件循环，等待 invokeMethod 触发任务）
    self._query_thread.start()
moveToThread 原理：Worker 对象被"移动"到工作线程后，所有通过 invokeMethod（QueuedConnection）调用的方法都在工作线程执行。但信号发出后，槽函数在接收者（MainWindow）所在的线程（即主线程）执行——这正是我们需要的：SQL 查询在工作线程，UI 更新在主线程。

4.4 改造 refresh_cover_view() —— 异步查询 + 缓存 （替换原第380~406行）

修改后：

def refresh_cover_view(self) -> None:
    """触发后台查询，若有有效缓存则直接使用"""
    import time

    # 缓存有效（5 秒内），直接渲染，不走数据库
    if self._books_cache is not None \
       and (time.time() - self._cache_timestamp) < 5:
        self._on_books_loaded(self._books_cache)
        return

    # 缓存过期或不存在 → 触发后台线程查询
    QMetaObject.invokeMethod(
        self._book_worker,
        "fetch_all_books",
        Qt.QueuedConnection     # 确保在工作线程的事件循环中执行
    )
为什么用 QMetaObject.invokeMethod 而不是直接调用？
直接调用 self._book_worker.fetch_all_books() 会在主线程执行——因为调用者（MainWindow）在主线程。必须通过 invokeMethod + QueuedConnection 将调用"排队"到工作线程的事件循环中，才能实现真正的异步。

4.5 新增方法：_on_books_loaded() —— 缓存 + 触发渐进式渲染 （插入到 refresh_cover_view 之后）

def _on_books_loaded(self, books: list):
    """收到后台查询结果（在主线程执行），缓存结果并启动渐进式渲染"""
    import time

    if not books:
        return

    # 更新缓存
    self._books_cache = books
    self._cache_timestamp = time.time()

    # 清除旧卡片
    self._clear_layout(self.cover_grid_layout)

    # 启动渐进式渲染
    self._render_cards_progressive(books)

def _on_query_error(self, error_msg: str):
    """查询出错时的处理"""
    Tips.information_msg(f"数据库查询失败：{error_msg}")

4.6 新增方法：_render_cards_progressive() + _render_next_batch() （插入到 _on_books_loaded 之后）

核心思路：每批创建 8 张卡片 → 用 QTimer.singleShot 延迟 16ms → 创建下一批。16ms 的间隔让 Qt 事件循环有机会处理用户输入、重绘界面，从而避免"假死"。

# 渐进式渲染配置
BATCH_SIZE = 8          # 每批创建的卡片数
BATCH_INTERVAL = 16     # 批次间隔 (ms)，约 60fps 一帧


def _render_cards_progressive(self, books: list):
    """初始化渐进式渲染状态，启动第一批"""
    self._pending_books = books
    self._pending_index = 0
    self._col_index = 0
    self._cols = 4               # 每行卡片数

    # 0ms 后立即执行第一批（在事件循环空闲时）
    QTimer.singleShot(0, self._render_next_batch)


def _render_next_batch(self):
    """创建下一批卡片（由 QTimer 回调，在主线程执行）"""
    books = self._pending_books
    cols = self._cols

    for _ in range(BATCH_SIZE):
        if self._pending_index >= len(books):
            # 全部渲染完成，清理临时状态
            self._pending_books = None
            self.statusbar.showMessage(
                self._translate("Views", f"封面视图已刷新，共 {len(books)} 本书")
            )
            return

        book = books[self._pending_index]

        # 提取书籍字段（索引对照数据库表结构）
        book_name = book[1] or "未知"
        author = book[4] or "未知"      # 修复：原代码误用索引 3
        book_type = book[11] or "未知"

        # 每 cols 本新建一水平行
        if self._col_index % cols == 0:
            row_layout = QHBoxLayout()
            row_layout.setSpacing(16)
            self.cover_grid_layout.addLayout(row_layout)

        # 创建卡片并加入当前行
        card = self._create_book_card(book_name, author, book_type)
        # 获取当前行的最后一个布局（即刚加的那个行布局）
        item = self.cover_grid_layout.itemAt(
            self.cover_grid_layout.count() - 1
        )
        if item and item.layout():
            item.layout().addWidget(card)

        self._pending_index += 1
        self._col_index += 1

    # 本批完成，延迟 BATCH_INTERVAL ms 后调度下一批
    QTimer.singleShot(BATCH_INTERVAL, self._render_next_batch)

4.7 新增 closeEvent() —— 安全清理线程 （插入到 MainWindow 类末尾、_translate 之前）

def closeEvent(self, event):
    """窗口关闭时安全退出后台线程"""
    self._query_thread.quit()      # 退出线程的事件循环
    self._query_thread.wait(3000)  # 等待线程结束（最多 3 秒）
    self.db.close()                # 关闭数据库连接
    event.accept()
为什么 quit() 之后必须 wait()？
如果只 quit() 不 wait()，线程可能还在运行就被强制终止，此时若持有锁或正在写文件会导致数据损坏。wait() 会阻塞直到线程完全退出（最多等待 3 秒超时保护）。

10. 常见问题

Q1: 为什么不能用 Python 的 threading.Thread？

可以用，但 Qt 的 QThread + moveToThread 提供了信号槽的跨线程自动排队机制。当 Worker 在工作线程发出 books_ready 信号时，Qt 会自动将槽函数 _on_books_loaded 的调用"排队"到主线程，无需手动加锁。用 threading.Thread 的话你需要自己把结果安全地传回主线程。

Q2: 工作线程查询和主线程写入会冲突吗？

每个查询都创建独立的 sqlite3 连接，SQLite 默认允许多个连接同时读。但如果有并发写入（扫描添加书籍），建议开启 WAL 模式：在 DatabaseManager.create_table() 中加一行 self.cursor.execute("PRAGMA journal_mode=WAL")。以当前的写入频率（手动触发）来说，不开 WAL 也基本安全。

Q3: 缓存 5 秒够吗？需要可配置吗？

5 秒是一个合理的起点：覆盖了"用户看完封面点回列表又立刻点回封面"的典型场景，又不会让缓存过期时间过长导致数据不一致。后续可以把这个值提取为 CACHE_TTL = 5 放到类常量中方便调整。

Q4: 如果书籍数量非常大（500+），渐进式渲染还会卡吗？

初始渲染不卡（因为有分批间隔），但 500+ 个 QFrame 全部创建后，在 QScrollArea 中滚动时 Qt 需要计算布局和重绘可见区域，可能有轻微掉帧。Qt 的 QScrollArea 默认只绘制可见控件，所以实际压力不大。如果确实有性能问题，后续可考虑 QListView + 自定义 Delegate 的虚拟化方案。

Q5: closeEvent 中 wait() 阻塞会导致窗口关闭变慢吗？

设置了 3 秒超时，大多数情况下线程在几毫秒内就能退出。3 秒是兜底保护，防止极端情况下无限等待。
